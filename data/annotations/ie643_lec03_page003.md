$$\tilde{w} = \begin{bmatrix} w_1 \\ w_2 \\ w_3 \\ \vdots \\ w_d \\ -\theta \end{bmatrix}_{d+1} \qquad \tilde{x} = \begin{bmatrix} x_1 \\ x_2 \\ \vdots \\ x_d \\ 1 \end{bmatrix}_{d+1}$$

$$\vec{w} = \begin{bmatrix} w_1 \\ w_2 \\ \vdots \\ w_d \end{bmatrix} \qquad \vec{x} = \begin{bmatrix} x_1 \\ x_2 \\ \vdots \\ x_d \end{bmatrix}$$

$$\sum_{j=1}^{d} w_j x_j \geq \theta \; ; \; < \theta \quad \text{(decision rule)}$$

$$\langle \vec{w}, \vec{x} \rangle - \theta \geq 0 \; ; \; < 0$$

$$\langle \tilde{w}, \tilde{x} \rangle \geq 0 \; ; \; < 0$$
